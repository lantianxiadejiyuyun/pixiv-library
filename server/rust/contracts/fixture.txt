CREATE TABLE jobs (
	id VARCHAR(36) NOT NULL,
	kind VARCHAR(32) NOT NULL,
	status VARCHAR(24) NOT NULL,
	progress FLOAT NOT NULL,
	payload JSON NOT NULL,
	queue_lane VARCHAR(24) NOT NULL GENERATED ALWAYS AS (CASE WHEN (coalesce(JSON_EXTRACT(payload, '$."retry_queue"'), 0) IS 1) THEN 'retry' WHEN (kind IN ('import', 'import_content') AND coalesce(JSON_EXTRACT(payload, '$."download_origin"'), '') = 'manual') THEN 'manual_download' WHEN (kind = 'import_content' OR kind = 'import' AND coalesce(JSON_EXTRACT(payload, '$."platform"'), '') = 'pixiv' AND (JSON_EXTRACT(payload, '$."download"') IS NULL OR JSON_EXTRACT(payload, '$."download"'))) THEN 'download' WHEN (kind = 'import') THEN 'details' WHEN (kind IN ('source_sync', 'series_sync', 'author_profile', 'author_feed', 'pixiv_bookmark')) THEN 'checks' WHEN (kind = 'ai') THEN 'ai' WHEN (kind = 'translate') THEN 'translation' ELSE 'other' END) VIRTUAL,
	download_account_id VARCHAR(36) GENERATED ALWAYS AS (CASE WHEN (length(JSON_EXTRACT(payload, '$."download_account_id"')) <= 36) THEN JSON_EXTRACT(payload, '$."download_account_id"') END) VIRTUAL,
	details_account_id VARCHAR(36) GENERATED ALWAYS AS (CASE WHEN (length(JSON_EXTRACT(payload, '$."details_account_id"')) <= 36) THEN JSON_EXTRACT(payload, '$."details_account_id"') END) VIRTUAL,
	check_account_id VARCHAR(36) GENERATED ALWAYS AS (CASE WHEN (length(JSON_EXTRACT(payload, '$."check_account_id"')) <= 36) THEN JSON_EXTRACT(payload, '$."check_account_id"') END) VIRTUAL,
	execution_account_id VARCHAR(36) GENERATED ALWAYS AS (CASE WHEN (length(JSON_EXTRACT(payload, '$."execution_account_id"')) <= 36) THEN JSON_EXTRACT(payload, '$."execution_account_id"') END) VIRTUAL,
	stats_details_account_id VARCHAR(36) GENERATED ALWAYS AS (CASE WHEN (details_account_id = download_account_id) THEN NULL ELSE details_account_id END) VIRTUAL,
	stats_check_account_id VARCHAR(36) GENERATED ALWAYS AS (CASE WHEN (check_account_id = download_account_id OR check_account_id = details_account_id) THEN NULL ELSE check_account_id END) VIRTUAL,
	stats_execution_account_id VARCHAR(36) GENERATED ALWAYS AS (CASE WHEN (execution_account_id = download_account_id OR execution_account_id = details_account_id OR execution_account_id = check_account_id) THEN NULL ELSE execution_account_id END) VIRTUAL,
	result JSON,
	error TEXT,
	attempts INTEGER NOT NULL,
	lease_until DATETIME,
	worker_id VARCHAR(64),
	created_at DATETIME NOT NULL,
	updated_at DATETIME NOT NULL,
	PRIMARY KEY (id)
);
